import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

llm-proxy is a self-hosted proxy for LLM inference APIs. It passes requests
through to your providers in their own format. It adds token metering,
virtual keys and JWT authentication, token limits, spend budgets, guardrails,
failover and weighted routing between providers, and usage records. The
1.x releases follow the [stability policy](/docs/llm-proxy/reference/stability/).

<CardGrid>
  <Card title="Passthrough, not translation" icon="right-arrow">
    Clients send OpenAI or Anthropic requests. The proxy forwards them in the
    same dialect. It does not convert between dialects.
  </Card>
  <Card title="Control who spends what" icon="approve-check">
    Virtual keys, JWT principals, per-tenant model access, token limits, and
    spend budgets in US dollars.
  </Card>
  <Card title="Strict configuration" icon="setting">
    An unknown or misspelled field is a load error with its full dotted path.
  </Card>
  <Card title="Per-provider credentials" icon="warning">
    Provider keys live on the provider, not on the route. A failover never
    sends one provider's key to another provider.
  </Card>
</CardGrid>
